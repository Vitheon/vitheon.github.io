### A. All Closed

不妨设 $T$ 为要添加的数的集合， $I$ 为所有 $S_i$ 的交集。那么有：

- $T \cap I = \varnothing$ 
- $T\cup I = T\cup (S_1 \cap S_2 \cap \dots \cap S_n) = \cap_{i = 1}^n T \cup S_i$ 

由于 $T\cup S_i$ 都是 线性空间，那么 线性空间的交 一定也是线性空间。即：$T\cup I$ 是线性空间。

不妨设 $U = T\cup I$ 。这里这么做的原因是：我们希望构造出最小的 $U$ 然后得到 $T = U - I$ ，这样得到的一定是最小的 $T$ 。

由于 $T \cup S_i$ 是线性空间，则有： $span(S_i) - S_i \subseteq U$ ，这里 $span(S_i)$ 表示 $S_i$ 的张成。所以我们可以据此把所有 $span(S_i)$ 贡献的数值填入到 $U$ 中。这里依据 $\mid span(S_i)  - S_i \mid$ 的大小来加速这个过程。

- 若有 $\mid span(S_i) - S_i \mid > \dfrac{1}{2} \mid span(S_i) \mid$ ，则 $span(S_i) - S_i$ 对应的线性空间一定是 $span(S_i)$ 。可以直接把 $S_i$ 中的所有数加入到 $U$ 中即可。复杂度为：$\mathcal O(c_i\times m)$ 
- 若有 $\mid span(S_i) - S_i \mid \leq \dfrac{1}{2} \mid span(S_i) \mid$，则有：$\mid span(S_i) \mid < 2\mid S_i \mid$ ，可以暴力的求出 $span(S_i)$ ，然后把 $S_i$ 中的数剔除。复杂度为：$\mathcal O(c_i\times m)$ 

这一步总的复杂度为：$\mathcal O(\sum c_i \times m)$ 。

由 $T \cup S_i = U\cup span(S_i)$ ，等式左右都应是线性空间，而由于 $U$ 和 $span(S_i)$ 都是线性空间，两个线性空间的并是线性空间，当且仅当 $U\subseteq span(S_i)$ 或 $U\supseteq span(S_i)$ 。

需要检查每一个 $span(S_i)$ 和 $U$ 的关系。注意到 $U$ 的基底最多变化 $m$ 次，考虑每当 $U$ 基地变化时，考虑新增加的基底能否被所有还没有被归类到 $U\supseteq span(S_i)$ 的 $span(S_i)$ 表达。如果不可以，则应该把 $S_i$ 归类到这一部分（后续把 $S_i$ 全部内容加入到 $U$ 的线性基中）。注意到，这样做（检查）的复杂度是 $\mathcal O(nm^2)$ ，如果每次当 $U$ 基底变化时候检查 $U$ 和 $S_i$ 的包含性，复杂度是 $\mathcal O(nm^3)$ 的。

最后把得到的 $U$ 中所有属于 $I$ 的数剔除就得到了 $T$ 。

在过程中，时刻维护 $U$ 的线性基即可（可以表达出所有 $U$ 中的数）。

		for(int i = n - 1; i >= 0; i --){
			int p = ((val >> i) & 1);
			if(p){
				if(!lb[i]){

			}

			}

		}

			int p = ((vv >> i) & 1);
			if(p && lb[i]) vv ^= lb[i];
		}
		return vv == 0;
	}

### H. Hidden Track (Easy Version)

注意到这个交互题的交互信息非常的奇怪，返回的是询问的答案模 $3$ 的结果。

考察询问的形式，可以支持反转一个数是否存在于查询集合中。进一步可以发现两次询问的差别如果仅仅只有一个数是否在询问集合中，那么通过交互器返回的结果可以知道该数相邻的数关于 $m$ 的关系。

具体的，考虑以所有 2 的幂次作为 $m$ ，然后 $v$ 取遍 $-1$ 和 $1\sim n$ 。询问次数一共为：$nk$ 。

对于特定的 $m’$ 和数 $b$ ，考察两次询问 $(m', -1)$ 和 $(m', b)$ 。其中只有一次查询集合中存在 $b$ ，考察从没有 $b$ 的询问到有 $b$ 的询问的结果的变化（此处假设 $b$ 不位于两边）。

- 如果 $b$ 相邻的数都不包含 $m'$ ，则答案 $+2$ 
- 如果 $b$ 相邻的数有一个包含 $m'$ ，则答案不变
- 如果 $b$ 相邻的数都包含 $m'$ ，则答案 $-2$

上述三种情况是可以在模 $3$ 意义下做出区分的。

但是如果 $b$ 位于两边（位置 $1$ 或者 位置 $n$ ）那么表现出来的效果就不同于上面的情况了，所以需要先找位于两边的数。

- 如果 $b$ 位于两边，如果 $b$ 相邻的数不包含 $m'$ ，则答案 $+1$
-  如果 $b$ 位于两边，如果 $b$ 相邻的数包含 $m'$ ，则答案 $-1$

注意到，如果 $b$ 位于两边，那么对于所有可能的 $m$ ，询问结果一定不会出现 答案不变的情况。若 $b$ 位于两边，那么一定会至少出现一次 答案不变（否则左右两边的数就相等了，与排列矛盾）。借助这个观察，可以得到两边的数是什么。

那么在处理完两边的数并且得到了所有上述询问的结果之后，我们可以推出 $v$ 取 0 的询问结果是什么。

之后可以从左向右依次推出每一位填什么（借助上面询问可以纪记录每个数旁边的数在每一个二进制位下 $1$ 的数量）。

复杂度：$\mathcal O(nk)$

#define pb emplace_back

typedef long long ll;
typedef double db;
typedef long double ldb;
typedef unsigned long long ull;
const int DMAX = 1000 + 5;
const double INF = 1e9;
const int MOD = 998244353;
using namespace std;
mt19937 rnd(chrono::steady_clock::now().time_since_epoch().count());
template<class T> inline void read(T &x){
	T f=1; x=0; char ch=getchar();
	while(ch<'0' || ch>'9'){
		if(ch=='-') f=-1; 
		ch=getchar();
	}
	while(ch<='9' && ch>='0'){ x=x*10+ch-'0';ch=getchar(); }
	x*=f;
}
int n;
int query(int m, int v){
	cout<<"? "<<m<<" "<<v<<endl;
	int x;
	read(x);
	if(x == -1) exit(0);
	return x;
}
int ans[DMAX][20], rep[20];
int near[DMAX][20];
int a[DMAX];
void solve(){
	read(n);
	if(n == 1){
		cout<<"! "<<0<<endl;
		return ;
	}
	int k = 0;
	while((1 << (k + 1)) <= n - 1) k ++;
	for(int j = 0; j <= k; j ++){
		rep[j] = query((1 << j), -1);
		for(int i = 1; i < n; i ++) ans[i][j] = query((1 << j), i);
	}
	int p1 = -1, p2 = -1;
	for(int i = 1; i < n; i ++){
		bool flag = 1;
		for(int j = 0; j <= k; j ++){
			if(ans[i][j] - rep[j] == 0) flag = 0;
		}
		if(flag){
			if(p1 == -1) p1 = i;
			else p2 = i;
		}
	}
	if(p2 == -1) p2 = 0;
	if(p1 > p2) swap(p1, p2);
	// cout<<p1<<" "<<p2<<endl;
	for(int j = 0; j <= k; j ++){
		int tot = 0;
		for(int i = 1; i < n; i ++){
			int vv = (i & (1 << j));
			int pop = __builtin_popcount(vv);
			int notin, in;
			if(pop & 1){
				in = rep[j], notin = ans[i][j];
				if(p1 == i || p2 == i){
					if((notin + 1) % 3 == in % 3) near[i][j] = 0, tot += 1;
					else near[i][j] = 1, tot += 0;
					continue;
				}
				if((notin + 2) % 3 == in % 3) near[i][j] = 0, tot += 2;
				if(notin == in) near[i][j] = 1, tot += 1;
				if((notin + 1) % 3 == in % 3) near[i][j] = 2, tot += 0;
			}
			else{
				in = ans[i][j], notin = rep[j];
				if(p1 == i || p2 == i){
					if((notin + 1) % 3 == in % 3) near[i][j] = 0, tot += 0;
					else near[i][j] = 1, tot += 1;
					continue;
				}
				if((notin + 2) % 3 == in % 3) near[i][j] = 0, tot += 0;
				if(notin == in) near[i][j] = 1, tot += 1;
				if((notin + 1) % 3 == in % 3) near[i][j] = 2, tot += 2;
			}
		}
		// cout<<j<<" "<<rep[j]<<" "<<tot<<endl;
		int diff = (2 * rep[j] - tot) % 3 + 3;
		diff %= 3;
		if(p1 == 0){
			if(diff == 0) near[0][j] = 0;
			else near[0][j] = 1;
		}
		else{
			if(diff == 2) near[0][j] = 2;
			if(diff == 1) near[0][j] = 1;
			if(diff == 0) near[0][j] = 0;
		}
	}
	// for(int i = 0; i <= k; i ++){
	// 	cout<<near[1][i]<<" ";
	// }
	// cout<<endl;
	a[1] = p1;
	for(int i = 1; i < n; i ++){
		int val = 0;
		for(int j = 0; j <= k; j ++){
			if(near[a[i]][j]) val |= (1 << j), near[a[i]][j] --;
		}
		for(int j = 0; j <= k; j ++){
			int p = ((a[i] >> j) & 1);
			if(p) near[val][j] --;
		}
		a[i + 1] = val;
	}
	// assert(p2 == a[n]);
	cout<<"! ";
	for(int i = 1; i <= n; i ++) cout<<a[i]<<" ";
	cout<<endl;
	return ;
}
int main(){
	int T;
	read(T);
	while(T--) solve();
	return 0;
}
